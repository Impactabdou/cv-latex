% --- STANDARD COMPONENTS ---
\newcommand{\resumeItem}[1]{%
  \setbox0=\hbox{#1}% Mesure la largeur du contenu (même s'il est dans une variable)
  \ifdim\wd0=0pt
    % Ne fait rien si le texte ou la variable est vide
  \else
    \item\fontsize{10}{10}\selectfont{#1}%
  \fi
}

\newcommand{\resumeSubheading}[4]{
  \vspace*{-1pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace*{-1pt}
}

\newcommand{\resumeProjectHeading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \small#1 & #2 \\
    \end{tabular*}\vspace{-2pt}
}

\renewcommand\labelitemi{$\vcenter{\hbox{\small$\bullet$}}$}
\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}[label={\small$\bullet$}]}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace*{-1pt}}


% --- NEW CUSTOM COMPONENTS (For Title: Content format) ---
\newcommand{\resumeInfoItem}[1]{%
  \setbox0=\hbox{#1}%
  \ifdim\wd0=0pt
    % Ne fait rien si le texte ou la variable est vide
  \else
    \vspace{-2pt}\item \small{#1}%
  \fi
}

\newcommand{\resumeReferenceItem}[3]{
  \vspace{-2pt}\item \small{\textbf{#1}{: #2} \\ #3} \vspace{-1pt}
}
